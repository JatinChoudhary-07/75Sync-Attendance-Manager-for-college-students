        function calcStats() {

                });
            });

        }

        function getForecast(p, a) {
            const t = p + a; if (t === 0) return { text: "Mark data to see forecast", type: "info" };

            else { const c = Math.floor((p - 0.75 * t) / 0.75); return { text: c > 0 ? `Safe to miss ${c} classes` : "Edge reached", type: "safe" }; }
        }

        // ACTIONS
        async function commit(idx, status) {
            if (!user) return; const updated = { ...history };
            if (!updated[currentSelectedDate]) updated[currentSelectedDate] = {};
            if (updated[currentSelectedDate][idx] === status) delete updated[currentSelectedDate][idx];
            else updated[currentSelectedDate][idx] = status;
            await sync(updated);
        }

        async function markFullDay(status) {
            if (!user) return; const day = new Date(currentSelectedDate).toLocaleDateString('en-US', { weekday: 'long' });
            const slots = timetable[day] || []; const updated = { ...history };
            if (!updated[currentSelectedDate]) updated[currentSelectedDate] = {};
            slots.forEach((_, i) => updated[currentSelectedDate][i] = status);
            await sync(updated);
        }
